namespace App\Http\Controllers;

class ProjectController
{
  public function deploy(Project $p)
  {
    $p->markAsLive();
    event(new ProjectDeployed($p));

    return response()->json([
      'status' => 'deployed',
      'url'    => $p->production_url,
    ]);
  }
}
